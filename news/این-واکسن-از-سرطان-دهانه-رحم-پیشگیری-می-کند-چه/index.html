این واکسن از سرطان دهانهٔ رحم پیشگیری می‌کند؛ چه کسانی در اروپا آن را می‌زنند؟

منبع: Euronews Persian • ۲۹ آبان • ۱۵:۴۱ • ۴ دقیقه مطالعه
داده‌های بهداشتی اتحادیه اروپا نشان می‌دهد که میزان واکسیناسیون HPV بر اساس کشور و جنسیت، در میان زنان و مردان کشورهای عضو، متفاوت است.

یک واکسن علیه یک ویروس شایع می‌تواند از چندین نوع سرطان پیشگیری کند، اما این که نوجوانان آن را دریافت می‌کنند یا نه تا حد زیادی به محل زندگی آن‌ها بستگی دارد؛ داده‌های سلامت اروپا چنین نشان می‌دهد.

بر اساس داده‌های مرکز اروپایی پیشگیری و کنترل بیماری‌ها (ECDC)، احتمال واکسینه شدن دختران نوجوان در کشورهای شمال اروپا، اسپانیا و پرتغال علیه ویروس پاپیلومای انسانی یا اچ‌پی‌وی (HPV) بیشتر از هر جای دیگر است، در حالی که پسران در اروپای مرکزی و شرقی کم‌ترین احتمال را برای دریافت این واکسن دارند.

اچ‌پی‌وی گروهی از ویروس‌های شایع است که برخی از آن‌ها از طریق رابطه جنسی منتقل می‌شوند. بیشتر عفونت‌ها خودبه‌خود برطرف می‌شوند، اما بعضی انواع می‌توانند در بدن باقی بمانند و سال‌ها بعد باعث سرطان شوند؛ از جمله سرطان دهانه رحم که هر سال حدود ۲۸۰ هزار زن را در سراسر جهان می‌کشد، و سرطان‌های نادرتر در زنان و مردان.

واکسن‌های اچ‌پی‌وی از سال ۲۰۰۶ میلادی در اتحادیه اروپا مجوز گرفته‌اند و بیشتر کشورها (منبع به زبان انگلیسی) توصیه می‌کنند که همه دختران و پسران حداکثر تا ۱۵ سالگی واکسینه شوند. افراد بزرگ‌تری که تاکنون واکسن نزده‌اند می‌توانند دوزهای جبرانی دریافت کنند.

مقام‌های بهداشتی مدت‌ها دو دوز واکسن را توصیه می‌کردند، اما ده‌ها کشور به برنامه واکسیناسیون تک‌دوزی روی آورده‌اند. پژوهش‌ها نشان می‌دهد که حتی یک نوبت تزریق هم ممکن است برای محافظت در برابر انواع اچ‌پی‌وی که بیشترین احتمال را برای ایجاد سرطان دارند کافی باشد.

کشورهایی که زودتر این واکسن را پذیرفتند، مانند سوئد و اسکاتلند (منبع به زبان انگلیسی)، در سال‌های بعد نرخ بسیار پایین‌تری از سرطان دهانه رحم گزارش کرده‌اند. با این حال، داده‌های این مرکز نشان می‌دهد که در عمل، در این که چه کسانی واکسن اچ‌پی‌وی را دریافت می‌کنند شکاف‌های بزرگی وجود دارد.

در پرتغال، ۹۵ درصد دختران تا سن ۱۵ سالگی همه دوزهای واکسن اچ‌پی‌وی را دریافت می‌کنند، در حالی که این نسبت در بلغارستان تنها ۴ درصد است. این نرخ‌ها در میان پسران به طور کلی پایین‌تر است، چون واکسن اچ‌پی‌وی برای آن‌ها فقط در سال‌های اخیر (منبع به زبان انگلیسی) پیشنهاد شده است. این میزان برای پسران در پرتغال ۹۴ درصد و در رومانی یک درصد گزارش شده است.

به طور میانگین، ۶۴ درصد دختران و ۵۱ درصد پسران تا ۱۵ سالگی همه دوزهای واکسن اچ‌پی‌وی را دریافت کرده بودند؛ رقمی که کمتر از هدف ۹۰ درصدی اتحادیه اروپا برای دختران (منبع به زبان انگلیسی) است.

چندین کشور برای افزایش ایمن‌سازی‌ها اقداماتی انجام داده‌اند. در سال ۲۰۲۱ میلادی، سوئد در چارچوب کارزار خود برای تبدیل شدن به نخستین کشوری که سرطان دهانه رحم را ریشه‌کن می‌کند، تزریق‌های جبرانی را برای زنانی که در جوانی واکسن نزده بودند آغاز کرد. تا سال ۲۰۲۵ میلادی، سوئد همچنین یکی از کوچک‌ترین شکاف‌های جنسیتی در واکسیناسیون اچ‌پی‌وی را داشت؛ به طوری که بر اساس داده‌های این مرکز، ۹۱ درصد دختران و ۸۷ درصد پسران تا ۱۵ سالگی واکسن را دریافت کرده بودند.

در همین حال، فرانسه در سال ۲۰۲۳ یک برنامه واکسیناسیون اچ‌پی‌وی مبتنی بر مدرسه را آغاز کرد (منبع به زبان انگلیسی)؛ در اولین سال اجرا، نرخ واکسیناسیون در میان ۱۲ ساله‌ها از ۳۱ درصد به ۴۸ درصد افزایش یافت. همچنین بلغارستان آغاز کرد (منبع به زبان انگلیسی) ارائه رایگان واکسن اچ‌پی‌وی به پسران را در سال ۲۰۲۵.

با این حال، به گفته مجموعه‌ای از مرورها که سال گذشته انجام شد (منبع به زبان انگلیسی) و توسط گروه پژوهشی غیردولتی کوکرین (Cochrane) (منبع به زبان انگلیسی) صورت گرفت، نرخ واکسیناسیون اچ‌پی‌وی در برخی کشورها کاهش یافته است که عمدتا به دلیل اطلاعات نادرست منتشرشده در شبکه‌های اجتماعی درباره این واکسن است.

یک مطالعه دیگر که در نشریه «بی‌ام‌جی پابلیک هلث» (منبع به زبان انگلیسی) منتشر شد، نشان داد در رومانی، بلغارستان و کرواسی، شایعاتی مبنی بر این که واکسن اچ‌پی‌وی ممکن است به کودکان آسیب برساند یا عوارض جانبی ایجاد کند، همراه با بی‌اعتمادی نسبت به مقام‌های بهداشتی، به توضیح این موضوع کمک می‌کند که چرا میزان استفاده از این واکسن پایین باقی مانده است.

به گفته کوکرین، که هم کارآزمایی‌های بالینی و هم مطالعات دنیای واقعی شامل بیش از ۱۳۲ میلیون نفر را تحلیل کرده است، هیچ مدرکی وجود ندارد که نشان دهد واکسن اچ‌پی‌وی باعث عوارض جانبی شایع گزارش‌شده می‌شود.

بر اساس این مرور، دختران نوجوانی که علیه اچ‌پی‌وی واکسینه شده بودند ۸۰ درصد کمتر از دختران واکسینه‌نشده در معرض ابتلا به سرطان دهانه رحم قرار داشتند.

پژوهشگران گفتند واکسن اچ‌پی‌وی زمانی بیشترین اثربخشی را دارد که افراد پیش از مواجهه با ویروس یا پیش از آغاز فعالیت جنسی آن را دریافت کنند.
